    for(int pos=timing;pos+sps<=(int)x.size();pos+=sps){
        double t=(pos+sps*.5)/FS;std::array<double,8>a{};int best=0;
        for(int s=0;s<8;s++){
            a[s]=goertzel(x,pos,sps,BASE_FREQ+s*p.spacing+center+drift*t);
            if(a[s]>a[best])best=s;
        }
        sy.push_back(best);pw.push_back(a);
    }
}

}
