    if(s.size()>6)throw std::runtime_error("call");

}
uint64_t enc_grid(const std::string&g){if(g.size()!=4)throw std::runtime_error("grid");int a=g[0]-'A',b=g[1]-'A',c=g[2]-'0',d=g[3]-'0';return (((a*18)+b)*10+c)*10+d;}
std::vector<std::string> split3(const std::string&s){std::vector<std::string>o;std::string cur;for(char ch:s){if(ch==' '){if(!cur.empty()){o.push_back(cur);cur.clear();}}else cur+=ch;}if(!cur.empty())o.push_back(cur);return o;}
std::vector<int> pack_message(const std::string&msg){auto p=split3(msg);if(p.size()!=3)throw std::runtime_error("msg parse");int type=0;uint64_t c1=0,c2=0,aux=0;if(p[0]=="CQ"){type=0;c1=enc_call(p[1]);aux=enc_grid(p[2]);}else if(p[2]=="RR73"){type=4;c1=enc_call(p[0]);c2=enc_call(p[1]);}else if(p[2]=="73"){type=5;c1=enc_call(p[0]);c2=enc_call(p[1]);}else if(p[2].size()>1&&p[2][0]=='R'){type=3;c1=enc_call(p[0]);c2=enc_call(p[1]);aux=std::stoi(p[2].substr(1))+50;}else if(p[2][0]=='+'||p[2][0]=='-'||std::isdigit((unsigned char)p[2][0])){type=2;c1=enc_call(p[0]);c2=enc_call(p[1]);aux=std::stoi(p[2])+50;}else{type=1;c1=enc_call(p[0]);c2=enc_call(p[1]);aux=enc_grid(p[2]);}std::vector<int>b;auto add=[&](std::vector<int>x){b.insert(b.end(),x.begin(),x.end());};add(i2bits(type,3));add(i2bits(c1,32));add(i2bits(c2,32));add(i2bits(aux,16));b.insert(b.end(),13,0);return b;}
std::vector<int> conv_encode(const std::vector<int>&bits){int st=0;std::vector<int>o;for(size_t i=0;i<bits.size()+TAIL;i++){int bit=i<bits.size()?bits[i]:0;int r=((st<<1)|bit)&127;o.push_back(parity(r&G0));o.push_back(parity(r&G1));st=r&63;}return o;}
std::vector<int> qcframe(const std::vector<int>&r){return qcenc(r);}
std::vector<int> exact_frame_symbols(const std::string&msg,const Profile&p){auto info=pack_message(msg);auto bytes=b2bytes(info,INFO_BITS);uint16_t crc=crc16(bytes);auto raw=info;auto cb=i2bits(crc,16);raw.insert(raw.end(),cb.begin(),cb.end());std::vector<int>coded;if(p.name=="WEAK")coded=qcframe(raw);else coded=conv_encode(raw);std::vector<int>rep;if(p.name=="WEAK")rep=coded;else for(int bit:coded)for(int r=0;r<p.repeat;r++)rep.push_back(bit);std::vector<int>bits;auto pid=i2bits(p.id,2);for(int bit:pid)for(int r=0;r<8;r++)bits.push_back(bit);bits.insert(bits.end(),rep.begin(),rep.end());while(bits.size()%3)bits.push_back(0);std::vector<int>data;for(size_t i=0;i<bits.size();i+=3)data.push_back((bits[i]<<2)|(bits[i+1]<<1)|bits[i+2]);int half=data.size()/2;std::vector<int>frame=SYNC;frame.insert(frame.end(),data.begin(),data.begin()+half);frame.insert(frame.end(),PILOT.begin(),PILOT.end());frame.insert(frame.end(),data.begin()+half,data.end());frame.insert(frame.end(),PILOT.begin(),PILOT.end());return frame;}
std::vector<double> reconstruct(const std::vector<double>&x,const Profile&p,const DecodeResult&r){int sps=int(FS/p.baud);std::vector<int>frame;try{frame=exact_frame_symbols(r.message,p);}catch(...){return std::vector<double>(x.size(),0.0);}int sampleStart=r.timing+r.index*sps;std::vector<double>bc(x.size(),0),bs(x.size(),0);double ph=0;int n=0;for(int tone:frame){for(int j=0;j<sps;j++){int idx=sampleStart+n;if(idx>=(int)x.size())break;double t=idx/double(FS);double f=BASE_FREQ+tone*p.spacing+r.center+r.drift*t;ph+=2*PI*f/FS;bc[idx]=std::cos(ph);bs[idx]=std::sin(ph);n++;}}double cc=0,ss=0,cs=0,xc=0,xs=0;for(size_t i=0;i<x.size();i++){cc+=bc[i]*bc[i];ss+=bs[i]*bs[i];cs+=bc[i]*bs[i];xc+=x[i]*bc[i];xs+=x[i]*bs[i];}double det=cc*ss-cs*cs+1e-12;double a=(xc*ss-xs*cs)/det,b=(xs*cc-xc*cs)/det;std::vector<double>y(x.size(),0);for(size_t i=0;i<y.size();i++)y[i]=a*bc[i]+b*bs[i];return y;}
std::vector<double> subtract(const std::vector<double>&x,const std::vector<double>&y){std::vector<double>o=x;for(size_t i=0;i<o.size()&&i<y.size();i++)o[i]-=y[i];return o;}
std::vector<DecodeResult> iterative_sic(const std::vector<double>&x,const Profile&p,int maxPasses=4){auto initial=wide(x,p);if(initial.empty())return{};std::vector<DecodeResult>all=initial;std::vector<DecodeResult>order=initial;std::sort(order.begin(),order.end(),[](auto&a,auto&b){return a.metric>b.metric;});std::vector<double>res=x;for(auto strong:order){auto model=reconstruct(res,p,strong);res=subtract(res,model);for(auto &weak:all){if(weak.message==strong.message&&std::abs(weak.center-strong.center)<20)continue;auto rr=local(res,p,weak.center);if(rr.ok&&rr.message==weak.message&&rr.metric>weak.metric)weak=rr;}}for(int pass=0;pass<maxPasses;pass++){auto more=wide(res,p);bool added=false;for(auto&r:more){bool dup=false;for(auto&e:all)if(same_frame(r,e,p))dup=true;if(!dup){all.push_back(r);res=subtract(res,reconstruct(res,p,r));added=true;}}if(!added)break;}std::vector<DecodeResult>out;for(auto&r:all){bool merged=false;for(auto&e:out){if(same_frame(r,e,p)){if(r.metric>e.metric)e=r;merged=true;break;}}if(!merged)out.push_back(r);}std::sort(out.begin(),out.end(),[](auto&a,auto&b){return a.center<b.center;});return out;}
const Profile&prof(const std::string&n){for(auto&p:PROFILES)if(p.name==n)return p;throw std::runtime_error("profile");}
