    for(double c=c0-5;c<=c0+5+1e-9;c+=fineStep){int tmin=std::max(0,t0-std::max(1,sps/8));int tmax=std::min(sps-1,t0+std::max(1,sps/8));for(int t=tmin;t<=tmax;t+=std::max(1,sps/32)){double m=sync_energy_score(x,p,t,bestIdx,c);if(m>bestM){bestM=m;bestC=c;bestT=t;}}}
    double bestD=0.0;for(int pass=0;pass<2;pass++){std::vector<FreqPoint>pts;auto add=[&](std::vector<FreqPoint>v){pts.insert(pts.end(),v.begin(),v.end());};add(known_region_points(x,p,bestT,bestIdx,SYNC,bestC,bestD));add(known_region_points(x,p,bestT,bestIdx+L.pilot1_start,PILOT,bestC,bestD));add(known_region_points(x,p,bestT,bestIdx+L.pilot2_start,PILOT,bestC,bestD));auto[df,dd]=weighted_line_fit(pts);if(std::abs(df)<=12)bestC+=df;if(std::abs(dd)<=1.5)bestD+=dd;}
    std::vector<int>sy;std::vector<std::array<double,8>>pw;symbol_powers(x,p,bestC,bestD,bestT,sy,pw);int s0=match_at(sy,SYNC,bestIdx);int s1=match_at(sy,PILOT,bestIdx+L.pilot1_start);int s2=match_at(sy,PILOT,bestIdx+L.pilot2_start);int minSync=((p.name=="WEAK"||p.name=="DEEP")?14:(p.name=="DX"?17:18));int minPilot=4;if(s0<minSync)return{};
    {double pc1=soft_pattern_confidence(pw,bestIdx+L.pilot1_start,PILOT);double pc2=soft_pattern_confidence(pw,bestIdx+L.pilot2_start,PILOT);if(p.name=="WEAK"||p.name=="DEEP"){if((pc1+pc2)<-0.45)return{};}}
    joint_ml_refine(x,p,bestIdx,bestT,bestC,bestD);symbol_powers(x,p,bestC,bestD,bestT,sy,pw);int syncScore=match_at(sy,SYNC,bestIdx);s1=match_at(sy,PILOT,bestIdx+L.pilot1_start);s2=match_at(sy,PILOT,bestIdx+L.pilot2_start);if(syncScore<minSync)return{};
    {double pc1=soft_pattern_confidence(pw,bestIdx+L.pilot1_start,PILOT);double pc2=soft_pattern_confidence(pw,bestIdx+L.pilot2_start,PILOT);if(p.name=="WEAK"||p.name=="DEEP"){if((pc1+pc2)<-0.45)return{};}}
    try{auto msg=decode_payload(strip_pilots(pw,bestIdx,p),p);double quality=known_symbol_ml_score(x,p,bestT,bestIdx,bestC,bestD);return{true,msg,p.name,syncScore,bestT,bestIdx,bestC,bestD,quality};}catch(...){return{};}
}

bool same_frame(const DecodeResult&a,const DecodeResult&b,const Profile&p){if(a.message!=b.message)return false;return std::abs(a.center-b.center)<=std::max(25.0,p.spacing*1.5)&&std::abs(a.timing-b.timing)<=int(FS/p.baud)/2;}
std::vector<DecodeResult>wide(const std::vector<double>&x,const Profile&p){auto cs=candidates(x,p);std::vector<std::future<DecodeResult>>fs;for(double c:cs)fs.emplace_back(std::async(std::launch::async,[&,c]{return local(x,p,c);}));std::vector<DecodeResult>raw;for(auto&f:fs){auto r=f.get();if(r.ok)raw.push_back(r);}std::vector<DecodeResult>out;for(auto&r:raw){bool merged=false;for(auto&e:out)if(same_frame(r,e,p)){if(r.metric>e.metric)e=r;merged=true;break;}if(!merged)out.push_back(r);}std::sort(out.begin(),out.end(),[](auto&a,auto&b){return a.center<b.center;});return out;}
